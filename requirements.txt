# Runtime dependencies for Currency Calc.
#
# Install into a fresh environment with:
#     python -m venv .venv
#     .venv\Scripts\pip install -r requirements.txt     (Windows)
#     .venv/bin/pip install -r requirements.txt        (macOS / Linux)
#
# Requires Python 3.9 or newer. main.py annotates with dict[str, str] and does
# not use "from __future__ import annotations", so those annotations are
# evaluated at runtime and 3.8 fails with
# "'type' object is not subscriptable". Verified: 3.9 is the first version that
# loads all three modules.
#
# Direct dependencies are pinned to the versions this was tested against
# (Python 3.14.7). Transitive dependencies are deliberately left unpinned so
# that security and bug fixes upstream flow in without a release here; pip
# resolves them, and Flask/Werkzeug/requests all respect the floor above.
#
# Only these two are needed. Do not add dependencies for things the standard
# library already covers.

flask==3.1.3
requests==2.34.2
